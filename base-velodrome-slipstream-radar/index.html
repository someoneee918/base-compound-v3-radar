import { ethers } from 'ethers';

const SLIPSTREAM_POOL = '0x8E8F5c48b264FA3FF66E40183bE7A6e4f32a76f2';
const BASE_RPC = 'https://mainnet.base.org';

const POOL_ABI = [
    'function slot0() view returns (uint160 sqrtPriceX96, int24 tick, uint16 observationIndex, uint16 observationCardinality, uint16 observationCardinalityNext, bool unlocked)',
    'function liquidity() view returns (uint128)'
];

async function monitorSlipstreamTick() {
    const provider = new ethers.JsonRpcProvider(BASE_RPC);
    const pool = new ethers.Contract(SLIPSTREAM_POOL, POOL_ABI, provider);
    console.log('[+] Monitoring Velodrome Slipstream Pool on Base (0x8E8F5c48b264FA3FF66E40183bE7A6e4f32a76f2)...');
    
    provider.on('block', async (blockNumber) => {
        try {
            const slot = await pool.slot0();
            console.log(`[Block ${blockNumber}] Current Tick: ${slot.tick}`);
        } catch (err) {}
    });
}
